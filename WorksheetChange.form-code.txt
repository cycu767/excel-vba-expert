' Paste into the intended worksheet code module, NOT a standard module or UserForm.
' Test-only example: uppercase literal text in B2:B1000, leave formulas/errors alone.
Option Explicit

Private Sub Worksheet_Change(ByVal Target As Range)
    Dim affected As Range
    Dim cell As Range
    Dim originalEvents As Boolean
    Dim captured As Boolean
    Dim number As Long
    Dim description As String
    Dim cleanupNumber As Long
    Dim cleanupDescription As String
    Dim value As Variant

    On Error GoTo Failed
    Set affected = Me.Application.Intersect(Target, Me.Range("B2:B1000"))
    If affected Is Nothing Then Exit Sub
    originalEvents = Me.Application.EnableEvents
    captured = True
    Me.Application.EnableEvents = False
    For Each cell In affected.Cells
        If Not cell.HasFormula Then
            value = cell.Value2
            If VarType(value) = vbString Then cell.Value2 = UCase$(CStr(value))
        End If
    Next cell
    GoTo CleanUp
Failed:
    number = Err.Number
    description = Err.Description
    Resume CleanUp
CleanUp:
    If captured Then
        On Error Resume Next
        Err.Clear
        Me.Application.EnableEvents = originalEvents
        cleanupNumber = Err.Number
        cleanupDescription = Err.Description
        On Error GoTo 0
    End If
    Set cell = Nothing
    Set affected = Nothing
    If number <> 0 Then
        If cleanupNumber <> 0 Then description = description & _
            " | Event-state restoration failed: " & cleanupDescription
        Err.Raise number, "Worksheet_Change", description
    End If
    If cleanupNumber <> 0 Then Err.Raise cleanupNumber, _
        "Worksheet_Change.Cleanup", cleanupDescription
End Sub
